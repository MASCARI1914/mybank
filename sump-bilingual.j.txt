  function createLangBtn(code) {
    var btn = document.createElement('button');
    btn.type = 'button';
    btn.className = 'sump-lang-btn';
    btn.setAttribute('data-lang-btn', code);
    btn.textContent = code.toUpperCase();

      window.setSumpLanguage(code);
    });
    return btn;
  }

    var boxes = document.querySelectorAll('.sump-topbar-right');
    for (var i = 0; i < boxes.length; i++) {
      var box = boxes[i];
      if (box.querySelector('.sump-lang-switcher')) continue;
      box.textContent = '';

      var wrap = document.createElement('div');
      wrap.className = 'sump-lang-switcher';

      var sep = document.createElement('span');
      sep.className = 'sump-lang-sep';
      sep.textContent = '|';

      wrap.appendChild(createLangBtn('el'));
      wrap.appendChild(sep);
      wrap.appendChild(createLangBtn('en'));
      box.appendChild(wrap);
    }
    updateLangButtonsUI();
  }

  function updateLangButtonsUI() {
    var btns = document.querySelectorAll('[data-lang-btn]');
    for (var i = 0; i < btns.length; i++) {
      var b = btns[i];
      if (b.getAttribute('data-lang-btn') === currentLang) {
        b.classList.add('is-active');
      } else {
        b.classList.remove('is-active');
      }
    }
  }

  function computeTranslation(raw) {
    var n = norm(raw);
    if (!n) return null;
    if (exactDict[n]) {
      return raw.replace(n, exactDict[n]);
    }
    if (!/[\u0370-\u03FF\u1F00-\u1FFF]/.test(raw)) return null;

    var out = raw;
    for (var i = 0; i < phrasePairs.length; i++) {
      var gr = phrasePairs[i][0];
      var en = phrasePairs[i][1];
      if (out.indexOf(gr) !== -1) {
        out = out.split(gr).join(en);
      }
    }
    return (out !== raw) ? out : null;
  }

  function translateTextNode(textNode) {
    var p = textNode.parentElement;
    if (!p) return;
    var tag = p.tagName;
    if (tag === 'SCRIPT' || tag === 'STYLE' || tag === 'NOSCRIPT') return;

    if (typeof textNode._origEL === 'undefined') {
      var raw = textNode.nodeValue;
      var tr = computeTranslation(raw);
      if (tr) {
        textNode._origEL = raw;
        textNode._transEN = tr;
      }
    }

    if (typeof textNode._origEL !== 'undefined') {
      var target = (currentLang === 'en') ? textNode._transEN : textNode._origEL;
      if (textNode.nodeValue !== target) {
        textNode.nodeValue = target;
      }
    }
  }

  function translateAttributes(el) {
    var attrs = ['placeholder', 'title', 'alt', 'aria-label'];
    for (var i = 0; i < attrs.length; i++) {
      var attr = attrs[i];
      if (!el.hasAttribute(attr)) continue;
      var storeKey = 'data-orig-el-' + attr;
      if (!el.hasAttribute(storeKey)) {
        var tr = computeTranslation(el.getAttribute(attr));
        if (tr) {
          el.setAttribute(storeKey, el.getAttribute(attr));
          el.setAttribute('data-trans-en-' + attr, tr);
        }
      }
      if (el.hasAttribute(storeKey)) {
        var nextVal = (currentLang === 'en')
          ? el.getAttribute('data-trans-en-' + attr)
          : el.getAttribute(storeKey);
        if (el.getAttribute(attr) !== nextVal) {
          el.setAttribute(attr, nextVal);
        }
      }
    }
  }

  function applyLanguageToTree(root) {
    root = root || document.body;
    if (!root) return;

    var walker = document.createTreeWalker(root, NodeFilter.SHOW_TEXT, null, false);
    var node;
    while ((node = walker.nextNode())) {
      translateTextNode(node);
    }

    if (root.querySelectorAll) {
      var allEls = root.querySelectorAll('[placeholder], [title], [alt], [aria-label]');
      for (var i = 0; i < allEls.length; i++) {
        translateAttributes(allEls[i]);
      }
    }
  }

  window.setSumpLanguage = function (lang) {
    if (lang !== 'el' && lang !== 'en') return;
    currentLang = lang;
    localStorage.setItem(STORAGE_KEY, lang);
    updateLangButtonsUI();
    applyLanguageToTree(document.body);
  };

  function initBilingual() {
    injectHeaderLangSwitcher();
    applyLanguageToTree(document.body);

    var observer = new MutationObserver(function (mutations) {
      for (var i = 0; i < mutations.length; i++) {
        var added = mutations[i].addedNodes;
        for (var j = 0; j < added.length; j++) {
          var n = added[j];
          if (n.nodeType === 1) applyLanguageToTree(n);
          else if (n.nodeType === 3) translateTextNode(n);
        }
      }
    });

    if (document.body) {
      observer.observe(document.body, { childList: true, subtree: true });
    }
  }

  if (document.readyState === 'loading') {
    document.addEventListener('DOMContentLoaded', initBilingual);
  } else {
    initBilingual();
  }
})();
